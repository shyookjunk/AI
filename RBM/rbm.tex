\section{Generative Models}
%---------------------------------------------------------------
\begin{itemize}
  \item Generative models are a class of machine learning models that can generate
        new data samples from the same distribution as the training data.
  \item The most important thing is to model the probability distribution
        that best represents the training data.
  \item They can be used for various applications, such as:
        \begin{itemize}
          \item Image generation
          \item Text generation
          \item Data augmentation
          \item Anomaly detection
        \end{itemize}
\end{itemize}

\section{Basic Mathematics}
\begin{itemize}
  \item Let $\mathbf{x}$ and $\mathbf{y}$ be the input and output vectors, respectively.
  \item Marginal distribution:
    \begin{derivbox}[Marginalization]
      \begin{equation}
        \label{eq:marginal}
        P(\mathbf{x}) = \sum_{\mathbf{y}} P(\mathbf{x}, \mathbf{y}).
      \end{equation}
    \end{derivbox}
  \item If we know the joint probability distribution $P(\mathbf{x}, \mathbf{y})$,
      we can derive the conditional distribution $P(\mathbf{y}|\mathbf{x})$
      using Bayes' theorem:
\end{itemize}
\begin{derivbox}[Bayes' Theorem]
  \begin{equation}
    \label{eq:bayes}
    P(\mathbf{y}|\mathbf{x}) = \frac{P(\mathbf{x}, \mathbf{y})}{P(\mathbf{x})}
                             = \frac{P(\mathbf{x}, \mathbf{y})}{\sum_{\mathbf{y^\prime}}P(\mathbf{x},\mathbf{y^\prime})}.
  \end{equation}
\end{derivbox}

\section{Restricted Boltzmann Machine (RBM)}
\begin{itemize}
  \item RBM is a type of probabilistic graph model that is used for unsupervised learning
        and feature extraction.
  \item RBM is a type of generative model that consists of two layers:
        a visible layer $\mathbf{v}$ and a hidden layer $\mathbf{h}$.
  \item The visible layer represents the input data, while the hidden layer
        captures the underlying features or patterns in the data.
  \item The connections between the visible and hidden layers are undirected,
        meaning that they can influence each other (see Fig.~\ref{fig:rbm}).
  \item RBM resembles the Ising model in statistical physics, where the visible and hidden units
        correspond to spins that can be in one of two states (0 or 1).
\end{itemize}

\subsection{Energy of RBM}
\begin{itemize}
  \item The energy function of an RBM for a given configuration of
  $\mathbf{v}=(v_1, v_2, \ldots, v_n)$ and
  $\mathbf{h}=(h_1, h_2, \ldots, h_m)$ is defined as:
\end{itemize}

\begin{equation}
  \label{eq:energy}
  \epsilon(\mathbf{v}, \mathbf{h}) = -\sum_{i,j} w_{ij} v_i h_j - \sum_i a_i v_i - \sum_j b_j h_j.
\end{equation}
In general, the weight between hidden and visible units $w_{ij}$ is {\bf{symmetric}}.

\section{Inference in RBM}

Since the RBM is a kind of probabilistic graphical model, the inference process
in RBM involves computing the joint probability of $\mathbf{v}$ and $\mathbf{h}$
and/or the probabilities to have certain configuration $\mathbf{v}$, $p(\mathbf{v})$.

Let's start with the joint probability distribution of $\mathbf{v}$ and $\mathbf{h}$.
From the statistical physics analogy, we can define the joint probability to have a
configuration $(\mathbf{v}, \mathbf{h})$ as a Boltzmann distribution:
\begin{equation}
  \label{eq:joint_prob}
  p(\mathbf{v}, \mathbf{h}) = \frac{e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{Z}
\end{equation}
where $Z$ is the partition function, defined as:
\begin{equation}
  \label{eq:partition_function}
  Z = \sum_{\mathbf{v}, \mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}.
\end{equation}

The probability of a visible vector $\mathbf{v}$ can be obtained by marginalizing over the hidden units:
\begin{samplebox}[Margenization to get $p(\mathbf{v})$]
  \begin{equation}
    \label{eq:visible_prob}
    p(\mathbf{v}) = \sum_{\mathbf{h}} p(\mathbf{v}, \mathbf{h})
    = \frac{1}{Z} \sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}
  \end{equation}
\end{samplebox}

Now, given a visible vector $\mathbf{v}$, we can compute the conditional probability of the hidden units:
\begin{equation}
  \label{eq:hidden_cond_prob}
  p(\mathbf{h}|\mathbf{v}) = \frac{p(\mathbf{v}, \mathbf{h})}{p(\mathbf{v})}
          =\prod_{j=1}^{m}p(h_j|\mathbf{v}).
\end{equation}
Similarly, we can compute the conditional probability of the visible units given the hidden units:
\begin{equation}
  \label{eq:visible_cond_prob}
  p(\mathbf{v}|\mathbf{h}) = \frac{p(\mathbf{v}, \mathbf{h})}{p(\mathbf{h})}
          =\prod_{i=1}^{n}p(v_i|\mathbf{h}).
\end{equation}
In Eqs.~\eqref{eq:hidden_cond_prob} and \eqref{eq:visible_cond_prob},
we used the fact that the hidden units are conditionally independent given the visible units,
and vice versa.
Note that the first equality in Eq.~\eqref{eq:hidden_cond_prob} can be rewritten as:
\begin{align}
   p(\mathbf{h}|\mathbf{v}) &= \frac{p(\mathbf{v}, \mathbf{h})}{p(\mathbf{v})} \nonumber \\
   &= \dfrac{\dfrac{e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{Z}}{\dfrac{1}{Z} \sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}} \nonumber \\
   &= \dfrac{e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{\sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}} \nonumber \\
   &= \dfrac{\exp\left(\sum_i\sum_j w_{ij} v_i h_j+\sum_i a_i v_i + \sum_j b_j h_j\right)}
   {\sum_{\mathbf{h}} \exp\left(\sum_i\sum_j w_{ij} v_i h_j+\sum_i a_i v_i + \sum_j b_j h_j\right)} \nonumber \\
   &= \dfrac{\exp\left(\sum_i a_i v_i\right)\exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_j b_j h_j\right)}
   {\exp\left(\sum_i a_i v_i\right)\sum_{\mathbf{h}} \exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_j b_j h_j\right)} \nonumber \\
   &= \dfrac{\exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_j b_j h_j\right)}
   {\sum_{\mathbf{h}} \exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_j b_j h_j\right)}.
   \label{eq:hidden_cond_prob_rewrite}
\end{align}
Similarly, the first equality in Eq.~\eqref{eq:visible_cond_prob} can be rewritten as:
\begin{align}
  p(\mathbf{v}|\mathbf{h}) &= \frac{p(\mathbf{v}, \mathbf{h})}{p(\mathbf{h})} \nonumber \\
  &= \dfrac{\dfrac{e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{Z}}{\dfrac{1}{Z} \sum_{\mathbf{v}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}} \nonumber \\
  &= \dfrac{e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{\sum_{\mathbf{v}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}} \nonumber \\
  &= \dfrac{\exp\left(\sum_i\sum_j w_{ij} v_i h_j+\sum_i a_i v_i + \sum_j b_j h_j\right)}
  {\sum_{\mathbf{v}} \exp\left(\sum_i\sum_j w_{ij} v_i h_j+\sum_i a_i v_i + \sum_j b_j h_j\right)} \nonumber \\
  &= \dfrac{\exp\left(\sum_j b_j h_j\right)\exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_i a_i v_i\right)}
  {\exp\left(\sum_j b_j h_j\right)\sum_{\mathbf{v}} \exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_i a_i v_i\right)} \nonumber \\
  &= \dfrac{\exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_j b_j h_j\right)}
  {\sum_{\mathbf{v}} \exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_j b_j h_j\right)}.
  \label{eq:visible_cond_prob_rewrite}
\end{align}

Eqs.~\eqref{eq:hidden_cond_prob_rewrite} and \eqref{eq:visible_cond_prob_rewrite} give us
the conditional probability that $h_j=1$ given $\mathbf{v}$
and the conditional probability that $v_i=1$ given $\mathbf{h}$, respectively:
\begin{align}
  p(h_j=1|\mathbf{v}) &= \dfrac{\exp\left(\sum_i w_{ij} v_i + b_j\right)
  \sum_{\mathbf{h^\prime}} \exp\left(\sum_i\sum_{k\neq j} w_{ik} v_i h_k + \sum_{k\neq j} b_k h_k\right)}
  {\sum_{\mathbf{h}} \exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_j b_j h_j\right)} \nonumber \\
  &= \dfrac{\exp\left(\sum_i w_{ij} v_i + b_j\right) \sum_{\mathbf{h'}} \exp\left(\sum_i\sum_{k\neq j} w_{ik} v_i h_k + \sum_{k\neq j} b_k h_k\right)}
   {\left[\exp\left(\sum_i w_{ij} v_i + b_j\right)+1\right] \sum_{\mathbf{h'}} \exp\left(\sum_i\sum_{k\neq j} w_{ik} v_i h_k + \sum_{k\neq j} b_k h_k\right)} \nonumber \\
   &= \dfrac{\exp\left(\sum_i w_{ij} v_i + b_j\right)}{\exp\left(\sum_i w_{ij} v_i + b_j\right) + 1} \nonumber \\
   &= \frac{1}{1 + \exp\left(-\sum_i w_{ij} v_i - b_j\right)} = \sigma\left(\sum_i w_{ij} v_i + b_j\right)
   \label{eq:hidden_cond_prob_final}
\end{align}
Here, $\sigma(x) = \frac{1}{1 + e^{-x}}$ is the sigmoid function and $\mathbf{h^\prime}$ denotes the configuration of hidden units except $h_j$.
Similarly, we can derive the conditional probability that $v_i=1$ given $\mathbf{h}$:
\begin{align}
  p(v_i=1|\mathbf{h}) &= \dfrac{\exp\left(\sum_j w_{ij} h_j + a_i\right)
  \sum_{\mathbf{v^\prime}} \exp\left(\sum_{k\neq i}\sum_j w_{kj} v_k h_j + \sum_{k\neq i} a_k v_k\right)}
  {\sum_{\mathbf{v}} \exp\left(\sum_i\sum_j w_{ij} v_i h_j + \sum_i a_i v_i\right)} \nonumber \\
  &= \dfrac{\exp\left(\sum_j w_{ij} h_j + a_i\right) \sum_{\mathbf{v'}} \exp\left(\sum_{k\neq i}\sum_j w_{kj} v_k h_j + \sum_{k\neq i} a_k v_k\right)}
   {\left[\exp\left(\sum_j w_{ij} h_j + a_i\right)+1\right] \sum_{\mathbf{v'}} \exp\left(\sum_{k\neq i}\sum_j w_{kj} v_k h_j + \sum_{k\neq i} a_k v_k\right)} \nonumber \\
   &= \dfrac{\exp\left(\sum_j w_{ij} h_j + a_i\right)}{\exp\left(\sum_j w_{ij} h_j + a_i\right) + 1} \nonumber \\
   &= \frac{1}{1 + \exp\left(-\sum_j w_{ij} h_j - a_i\right)} = \sigma\left(\sum_j w_{ij} h_j + a_i\right).
   \label{eq:visible_cond_prob_final}
\end{align}
Here, $\mathbf{v^\prime}$ denotes the configuration of visible units except $v_i$.

\subsection{Learning in RBM}

Let training dataset $V$ be composed of $M$ samples,
i.e., $V = \{\mathbf{v}_{1}, \mathbf{v}_{2}, \ldots, \mathbf{v}_{M}\}$.
Then likelihood function of the training dataset, $p(V)$, is defined as:
\begin{equation}
  \label{eq:likelihood}
  p(V) \equiv \prod_{m=1}^{M} p(\mathbf{v}_m).
\end{equation}
The \textcolor{red}{log-likelihood function} is given by:
\begin{checkpointbox}[Log-likelihood function]
\begin{equation}
  \label{eq:log_likelihood}
  \mathcal{L} \equiv \log p(V) = \sum_{m=1}^{M} \log p(\mathbf{v}_m)
  \equiv \sum_{m=1}^{M} \mathcal{L}_{\mathbf{v}_m}.
\end{equation}
where $\mathcal{L}_{\mathbf{v}_m} = \log p(\mathbf{v}_m)$ is the log-likelihood of a single
training sample $\mathbf{v}_m$.
\end{checkpointbox}

\begin{itemize}
  \item The learning process in RBM involves maximizing the log-likelihood
    function $\mathcal{L}$ with respect to the model parameters $\theta = \{w_{ij}, a_i, b_j\}$.
  \item This is typically done using gradient ascent, where we compute
    the gradients of $\mathcal{L}$ with respect to the parameters and update them iteratively.
  \item The gradients can be computed using the following formulas:
\end{itemize}

\begin{cautionbox}[Gradient Ascend Update]
  \begin{align}
    w_{ij}^{(t+1)} &= w_{ij}^{(t)} + \eta \frac{\partial \mathcal{L}}{\partial w_{ij}} \label{eq:update_w} \\
    a_i^{(t+1)} &= a_i^{(t)} + \eta \frac{\partial \mathcal{L}}{\partial a_i} \label{eq:update_a} \\
    b_j^{(t+1)} &= b_j^{(t)} + \eta \frac{\partial \mathcal{L}}{\partial b_j} \label{eq:update_b}
  \end{align}
\end{cautionbox}

When we use a single training sample $\mathbf{v}$, the gradients can be computed as follows:
\begin{align}
  \frac{\partial \mathcal{L}}{\partial w_{ij}} &= \dfrac{\partial \mathcal{L}_{\mathbf{v}}}{\partial w_{ij}} \nonumber \\
  &= \dfrac{\partial}{\partial w_{ij}} \ln p(\mathbf{v}) \nonumber \\
  &= \dfrac{\partial}{\partial w_{ij}} \ln \left[\dfrac{\sum_{\mathbf{h}}
        e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{Z}\right] \nonumber \\
  &= \dfrac{\partial}{\partial w_{ij}} \ln \sum_{\mathbf{h}}
        e^{-\epsilon(\mathbf{v}, \mathbf{h})}
        - \dfrac{\partial  \ln Z}{\partial w_{ij}} \nonumber \\
  &= \dfrac{1}{\sum_{\mathbf{h^\prime}} e^{-\epsilon(\mathbf{v}, \mathbf{h^\prime})}}
        \sum_{\mathbf{h}} \dfrac{\partial e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{\partial w_{ij}}
        - \dfrac{1}{Z} \dfrac{\partial Z}{\partial w_{ij}} \nonumber \\
  &= \dfrac{1}{\sum_{\mathbf{h^\prime}} e^{-\epsilon(\mathbf{v}, \mathbf{h^\prime})}}
        \sum_{\mathbf{h}} \dfrac{\partial e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{\partial w_{ij}}
        +\dfrac{1}{Z} \sum_{\mathbf{v}, \mathbf{h}} e^{-\epsilon(\mathbf{v},\mathbf{h})}
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial w_{ij}} \nonumber \\
  &=-\dfrac{1}{\sum_{\mathbf{h^\prime}} e^{-\epsilon(\mathbf{v}, \mathbf{h^\prime})}}
        \sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial w_{ij}}
        + \sum_{\mathbf{v}, \mathbf{h}} \textcolor{magenta}{\dfrac{e^{-\epsilon(\mathbf{v},\mathbf{h})}}{Z}}
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial w_{ij}} \nonumber \\
  &=-\sum_{\mathbf{h}} \textcolor{red}{\dfrac{e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{
                      \sum_{\mathbf{h^\prime}} e^{-\epsilon(\mathbf{v}, \mathbf{h^\prime})}}}
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial w_{ij}}
        + \sum_{\mathbf{v}, \mathbf{h}} \textcolor{magenta}{p(\mathbf{v},\mathbf{h})}
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial w_{ij}} \nonumber \\
  &=-\sum_{\mathbf{h}} \textcolor{red}{p(\mathbf{h}|\mathbf{v})}
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial w_{ij}}
        + \sum_{\mathbf{v}, \mathbf{h}} p(\mathbf{v},\mathbf{h})
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial w_{ij}} \nonumber \\
  &=-\sum_{\mathbf{h}} p(\mathbf{h}|\mathbf{v}) (-v_i h_j)
        + \sum_{\mathbf{v}, \mathbf{h}} p(\mathbf{v},\mathbf{h}) (-v_i h_j) \nonumber \\
  &=\sum_{\mathbf{h}} p(\mathbf{h}|\mathbf{v}) v_i h_j
        - \sum_{\mathbf{v}, \mathbf{h}} p(\mathbf{v},\mathbf{h}) v_i h_j \label{eq:gradient_w}
\end{align}
Here, we use the definition of the energy function $\epsilon(\mathbf{v}, \mathbf{h})$ in Eq.~\eqref{eq:energy},
and Eqs.~\eqref{eq:joint_prob} and \eqref{eq:hidden_cond_prob} in the derivation.
Since $h_i$ can take a value $0$ or $1$, the first term in the last line of Eq.~\eqref{eq:gradient_w}
is rewritten as:
\begin{align}
  \sum_{\mathbf{h}} p(\mathbf{h}|\mathbf{v}) v_i h_j &
    = \dfrac{\sum_{\mathbf{h}} \exp\left(\sum_{i,j}w_{ij}v_i h_j+\sum_j b_j h_j\right) v_i h_j}
          {\sum_{\mathbf{h}} \exp\left(\sum_{i,j}w_{ij}v_i h_j+\sum_j b_j h_j\right)} \nonumber \\
    &= \dfrac{\left\{\exp\left(\sum_i w_{ij} v_i+b_i\right) v_i +0\right\}
      \sum_{\mathbf{h}^\prime} \exp\left(\sum_{i,j}w_{ij}v_i h^\prime_j+\sum_j b_j h^\prime_j\right) v_i h^\prime_j}
          {\left\{\exp\left(\sum_i w_{ij} v_i+b_i\right) + 1\right\}
                  \sum_{\mathbf{h}^\prime} \exp\left(\sum_{i,j}w_{ij}v_i h^\prime_j+\sum_j b_j h^\prime_j\right)
                          v_i h^\prime_j} \nonumber \\
    &= \dfrac{1}{1 + \exp\left(-\sum_i w_{ij} v_i - b_j\right)} v_i \nonumber \\
    &= p(h_j=1|\mathbf{v}) v_i. \label{eq:gradient_w_first_term}
\end{align}
Therefore, the gradient of the log-likelihood with respect to $w_{ij}$ is given by:
\begin{align}
   \frac{\partial \mathcal{L}}{\partial w_{ij}} &= \dfrac{\partial \mathcal{L}_{\mathbf{v}}}{\partial w_{ij}} \nonumber \\
    &= \sum_{\mathbf{h}} p(\mathbf{h}|\mathbf{v}) v_i h_j
        - \sum_{\mathbf{v}, \mathbf{h}} p(\mathbf{v},\mathbf{h}) v_i h_j \nonumber \\
    &= p(h_j=1|\mathbf{v}) v_i - \sum_{\mathbf{v}, \mathbf{h}} p(\mathbf{v},\mathbf{h}) v_i h_j \nonumber \\
    &= \left<v_i h_j\right>_{\text{data}} - \left<v_i h_j\right>_{\text{model}}. \label{eq:gradient_w_final}
\end{align}
Therefore, we obtain $\left<v_i h_j\right>_{\text{data}}$ from the training data by just calculating
the sigmoid function in Eq.\eqref{eq:hidden_cond_prob_final}.

Similarly, we can derive the gradients with respect to $a_i$ and $b_j$:
\begin{align}
  \frac{\partial \mathcal{L}}{\partial a_i} &= \dfrac{\partial \mathcal{L}_{\mathbf{v}}}{\partial a_i} \nonumber \\
  &= \dfrac{\partial}{\partial a_i} \ln p(\mathbf{v}) \nonumber \\
  &= \dfrac{\partial}{\partial a_i} \ln \left[\dfrac{\sum_{\mathbf{h}}
        e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{Z}\right] \nonumber \\
  &= \dfrac{\partial}{\partial a_i} \ln \sum_{\mathbf{h}}
        e^{-\epsilon(\mathbf{v}, \mathbf{h})}
        - \dfrac{\partial  \ln Z}{\partial a_i} \nonumber \\
  &= \dfrac{1}{\sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}}
      \sum_{\mathbf{h}} \dfrac{\partial e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{\partial a_i}
        - \dfrac{1}{Z} \dfrac{\partial Z}{\partial a_i} \nonumber \\
  &= -\dfrac{1}{\sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}}
      \sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})} \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial a_i}
        + \dfrac{1}{Z} \sum_{\mathbf{v}, \mathbf{h}} e^{-\epsilon(\mathbf{v},\mathbf{h})}
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial a_i} \nonumber \\
  &= \sum_{\mathbf{h}} \dfrac{e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{\sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}} v_i
        - \sum_{\mathbf{v}, \mathbf{h}} \dfrac{e^{-\epsilon(\mathbf{v},\mathbf{h})}}{Z} v_i \nonumber \\
  &= \sum_{\mathbf{h}} p(\mathbf{h}|\mathbf{v}) v_i - \sum_{\mathbf{v}, \mathbf{h}} p(\mathbf{v},\mathbf{h}) v_i \nonumber \\
  &= v_i - \sum_{\mathbf{v}, \mathbf{h}} p(\mathbf{v},\mathbf{h}) v_i \nonumber \\
  &=\left<v_i\right>_{\text{data}} - \left<v_i\right>_{\text{model}}, \label{eq:gradient_a_final}
\end{align}
and
\begin{align}
  \frac{\partial \mathcal{L}}{\partial b_j} &= \dfrac{\partial \mathcal{L}_{\mathbf{v}}}{\partial b_j} \nonumber \\
  &= \dfrac{\partial}{\partial b_j} \ln p(\mathbf{v}) \nonumber \\
  &= \dfrac{\partial}{\partial b_j} \ln \left[\dfrac{\sum_{\mathbf{h}}
        e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{Z}\right] \nonumber \\
  &= \dfrac{\partial}{\partial b_j} \ln \sum_{\mathbf{h}}
        e^{-\epsilon(\mathbf{v}, \mathbf{h})}
        - \dfrac{\partial  \ln Z}{\partial b_j} \nonumber \\
  &= \dfrac{1}{\sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}} \sum_{\mathbf{h}}
        \dfrac{\partial e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{\partial b_j}
        - \dfrac{1}{Z} \dfrac{\partial Z}{\partial b_j} \nonumber \\
  &= -\dfrac{1}{\sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}} \sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}
      \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial b_j}
        + \dfrac{1}{Z} \sum_{\mathbf{v}, \mathbf{h}} e^{-\epsilon(\mathbf{v},\mathbf{h})}
        \dfrac{\partial \epsilon(\mathbf{v}, \mathbf{h})}{\partial b_j} \nonumber \\
  &= \sum_{\mathbf{h}} \dfrac{e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{\sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}} h_j
        - \sum_{\mathbf{v}, \mathbf{h}} \dfrac{e^{-\epsilon(\mathbf{v},\mathbf{h})}}{Z} h_j \nonumber \\
  &= \textcolor{red}{\sum_{\mathbf{h}} p(\mathbf{h}|\mathbf{v}) h_j} - \sum_{\mathbf{v}, \mathbf{h}}
      p(\mathbf{v},\mathbf{h}) h_j \nonumber \\
  &= \textcolor{red}{p(h_j=1|\mathbf{v})} - \sum_{\mathbf{v}, \mathbf{h}} p(\mathbf{v},\mathbf{h}) h_j \nonumber \\
  &=\left<h_j\right>_{\text{data}} - \left<h_j\right>_{\text{model}}.  \label{eq:gradient_b_final}
\end{align}
Here, we use again the fact that $h_j$ can take a value $0$ or $1$
in the derivation of the first term in the last line of Eq.~\eqref{eq:gradient_b_final}.

\begin{itemize}
  \item $\left<\cdots\right>_{\text{data}}$ denotes the expectation with respect to the data distribution.
  \item $\left<\cdots\right>_{\text{model}}$ denotes the expectation with respect to the model distribution.
  \item $\left<v_i h_j\right>_{\text{data}}$ can be computed using the training data and the conditional
    probability $p(h_j=1|\mathbf{v})$.
  \item $\left<v_i h_j\right>_{\text{model}}$ should be obtained from $\sum_{\mathbf{v}} p(\mathbf{v},\mathbf{h})$
    or equivalently from the model's marginal distribution, $\sum_{\mathbf{v}} p(\mathbf{v}) p(h_j=1|\mathbf{v})$,
    which is intractable to compute exactly due to the partition function $Z$,
    \begin{itemize}
      \item If the dimension of $\mathbf{v}$ is given by $n$, then we need to compute $2^n$ cases in the
      sum $\sum_{\mathbf{v}} p(h_j=1|\mathbf{v})$.
    \end{itemize}
\end{itemize}

\subsection{Kullback-Leibler (KL) Divergence}
In general, the learning process in RBM can be viewed as minimizing the Kullback-Leibler (KL) divergence
between the data distribution $p_{\text{data}}(\mathbf{v})$ and the model distribution $p_{\text{model}}(\mathbf{v})$:
\begin{equation}
  \label{eq:kl_divergence}
  D_{\text{KL}}(p_{\text{data}} || p_{\text{model}}) = \sum_{\mathbf{v}} p_{\text{data}}(\mathbf{v}) \log \frac{p_{\text{data}}(\mathbf{v})}{p_{\text{model}}(\mathbf{v})}.
\end{equation}
Minimizing the KL divergence is equivalent to maximizing the log-likelihood function $\mathcal{L}$, since:
\begin{align}
  D_{\text{KL}}(p_{\text{data}} || p_{\text{model}}) &= \sum_{\mathbf{v}} p_{\text{data}}(\mathbf{v}) 
              \log \frac{p_{\text{data}}(\mathbf{v})}{p_{\text{model}}(\mathbf{v})} \nonumber \\
          &= \sum_{\mathbf{v}} p_{\text{data}}(\mathbf{v}) \log p_{\text{data}}(\mathbf{v}) - 
              \sum_{\mathbf{v}} p_{\text{data}}(\mathbf{v}) \log p_{\text{model}}(\mathbf{v}) \nonumber \\
          &= -H(p_{\text{data}}) - \sum_{\mathbf{v}} p_{\text{data}}(\mathbf{v}) 
          \log p_{\text{model}}(\mathbf{v}) \nonumber \\
          &= -H(p_{\text{data}}) - \mathcal{L},
\end{align}
where $H(p_{\text{data}})$ is the entropy of the data distribution,
which is constant with respect to the model parameters. 
Therefore, minimizing $D_{\text{KL}}(p_{\text{data}} || p_{\text{model}})$ is equivalent 
to maximizing $\mathcal{L}$. This perspective helps us understand that the learning 
process in RBM is essentially about making the model distribution as close as possible 
to the data distribution by adjusting the model parameters to minimize the KL divergence.


\subsection{Markov Chain Monte Carlo (MCMC)}

\begin{itemize}
  \item In RBM, the value of each node changes stochastically.
  \item Use Markov Chain Monte Carlo (MCMC) methods, such as Gibbs sampling,
    to approximate the probability $\sum_{\mathbf{v}} p(\mathbf{v}) p(h_j=1|\mathbf{v})$.
\end{itemize}

In Fig.~\ref{fig:rbm}, for given values of $h_2,h_3,v_1,v_2,v_3,v_4$,
the probability of sampling $h_1$ is
\begin{equation}
  h_1\sim p(h_1|h_2,h_3,v_1,v_2,v_3,v_4).
  \label{eq:gibbs_sampling}
\end{equation}
In RBM, there is no connection between the nodes in the same layer (hidden layer),
so the value of $h_1$ is independent of $h_2$ and $h_3$ given the values of $v_1,v_2,v_3,v_4$.
Therefore, we can rewrite Eq.~\eqref{eq:gibbs_sampling} as:
\begin{equation}
  h_1\sim p(h_1|v_1,v_2,v_3,v_4).
\end{equation}
Since $h_1$ can take a value $0$ or $1$, we can compute the probability of sampling $h_1=1$, i.e.,
$p(h_1=1|v_1,v_2,v_3,v_4)$, using Eq.~\eqref{eq:hidden_cond_prob_final},
and set $h_1=1$ with that probability and $h_1=0$ with the complementary probability, in practice.
Therefore, we can sample the data of the hidden layer $\mathbf{h}=(h_1,h_2,h_3)$ as follows:
\begin{align}
  h_1 &\sim p(h_1|v_1,v_2,v_3,v_4) \nonumber \\
  h_2 &\sim p(h_2|v_1,v_2,v_3,v_4) \nonumber \\
  h_3 &\sim p(h_3|v_1,v_2,v_3,v_4) \nonumber  \\
  \mathbf{h} &= (h_1,h_2,h_3). \label{eq:sample_hidden_layer}
\end{align}

Similarly, we can sample the data of the visible layer $\mathbf{v}=(v_1,v_2,v_3,v_4)$
given the values of the hidden layer $\mathbf{h}=(h_1,h_2,h_3)$ as follows:
\begin{align}
  v_1 &\sim p(v_1|h_1,h_2,h_3) \nonumber \\
  v_2 &\sim p(v_2|h_1,h_2,h_3) \nonumber \\
  v_3 &\sim p(v_3|h_1,h_2,h_3) \nonumber  \\
  v_4 &\sim p(v_4|h_1,h_2,h_3) \nonumber  \\
  \mathbf{v} &= (v_1,v_2,v_3,v_4). \label{eq:sample_visible_layer}
\end{align}

\begin{itemize}
  \item \textbf{Burn-in algorithm}: Start with a training sample $\mathbf{v}$,
    sample $\mathbf{h}$ from $p(\mathbf{h}|\mathbf{v})$,
    then sample $\mathbf{v'}$ from $p(\mathbf{v}|\mathbf{h})$,
    and repeat this process until the probability distribution of the samples converges.
  \item \textbf{Contrastive Divergence (CD):} Start with a training sample $\mathbf{v}$,
    sample $\mathbf{h}$ from $p(\mathbf{h}|\mathbf{v})$,
    then sample $\mathbf{v}^\prime$ from $p(\mathbf{v}^\prime|\mathbf{h})$,
    and use $\mathbf{v}^\prime$ to compute the gradient of the
    log-likelihood function, \textit{without burn-in procedure}.
  \item \textbf{$k$-step CD ($CD_k$):} Start with a training sample $\mathbf{v}$,
    sample $\mathbf{h}$ from $p(\mathbf{h}|\mathbf{v})$,
    then sample $\mathbf{v}^\prime$ from $p(\mathbf{v}^\prime|\mathbf{h})$,
    and repeat this process $k$ times,
    and use the final $\mathbf{v}^\prime$ to compute
    the gradient of the log-likelihood function (see Fig.~\ref{fig:rbm_k_cd}).
\end{itemize}

  \textbf{Input}: Traning dataset $V = \{v_1, v_2, \ldots, v_M\}$, learning rate $\eta$ \\
  \textbf{Output}: Weight vector $\boldsymbol{W}$, bias vectors $a$, $b$

  \begin{enumerate}[leftmargin=1.5em, label=\arabic*.]
    \item Initialize $\boldsymbol{W}$, $a$, $b$ with small random values.
    \item \textbf{repeat}
    \item \quad \textbf{for} ( each $v$ in $V$) \{
    \item \quad\quad $v^0 \leftarrow v$
    \item \quad\quad \textbf{for} ($t = 1$ to $k$) \{
    \item \quad\quad\quad Compute $p(h_j^{k-1} = 1 | v^{k-1})$ using Eq. \eqref{eq:hidden_cond_prob_final}
                          and sample $\mathbf{h}^{k-1}$.
    \item \quad\quad\quad Compute $p(v_i^k = 1 | h^{k-1})$ using Eq. \eqref{eq:visible_cond_prob_final}
                          and sample $\mathbf{v}^k$.
    \item \quad\quad \}
    \item \quad\quad Update the weights $\left\{w_{ij}\right\}$ using Eq. \eqref{eq:gradient_w_final}.
    \item \quad\quad Update the biases $\left\{a_i\right\}$ using Eq. \eqref{eq:gradient_a_final}.
    \item \quad\quad Update the biases $\left\{b_j\right\}$ using Eq. \eqref{eq:gradient_b_final}.
    \item \quad \}
    \item \textbf{until} (Termination condition is met, e.g., number of epochs or convergence)
  \end{enumerate}
\end{algorithmbox}

\section{Thermodynamics and Statistical Mechanics Approach to RBM}
From Eq.\eqref{eq:visible_prob}, we can define the free energy of the visible layer $\mathbf{v}$ as follows:
\begin{checkpointbox}[Free Energy of the Visible Layer]
  \begin{equation}
    \label{eq:free_energy_def}
    F(\mathbf{v}) \equiv -\ln \sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}.
  \end{equation}
\end{checkpointbox}
Then the partition function $Z$ in Eq. \eqref{eq:visible_prob} can be rewritten as:
\begin{equation}
  \label{eq:partition_function_free_energy}
  Z=\sum_{\mathbf{v}}\sum_{\mathbf{h}}e^{-\epsilon(\mathbf{v}, \mathbf{h})}
  = \sum_{\mathbf{v}} e^{-F(\mathbf{v})}.
\end{equation}
From Eqs. \eqref{eq:visible_prob} and \eqref{eq:free_energy_def},
we can express the probability of the visible layer $\mathbf{v}$ as follows:
\begin{checkpointbox}[Visible Layer Probability in Terms of Free Energy]
  \begin{equation}
    \label{eq:visible_prob_free_energy}
    p(\mathbf{v}) = \dfrac{e^{-F(\mathbf{v})}}{\sum_{\mathbf{v}}\sum_{\mathbf{h}}
                                      e^{-\epsilon(\mathbf{v}, \mathbf{h})}}
              = \dfrac{e^{-F(\mathbf{v})}}{\sum_{\mathbf{v}} e^{-F(\mathbf{v})}}
              = \dfrac{e^{-F(\mathbf{v})}}{Z}.
  \end{equation}
\end{checkpointbox}

Let us define the biases and weight in the matrix form as follows:
\begin{itemize}
  \item $\mathbf{W}$ is the weight matrix between the visible and hidden layers.
  \item $\mathbf{a}$ is the bias vector for the visible layer.
  \item $\mathbf{b}$ is the bias vector for the hidden layer.
\end{itemize}
{\textcolor{red}{\bf{For symmetric $\mathbf{W}$,}}} the energy $\epsilon(\mathbf{v}, \mathbf{h})$ 
in Eq.\eqref{eq:energy} can be rewritten as:
\begin{equation}
  \label{eq:energy_matrix_form}
  \epsilon(\mathbf{v}, \mathbf{h}) = -\mathbf{a}^T \mathbf{v} - \mathbf{b}^T \mathbf{h}
    -\mathbf{v}^T \mathbf{W} \mathbf{h}
    =-\mathbf{a}^T \mathbf{v} - \mathbf{b}^T \mathbf{h}
    -\mathbf{h}^T \mathbf{W} \mathbf{v}.
\end{equation}

\subsection{Two State RBM: Bernoulli RBM}
\begin{itemize}
  \item $h_i\in \left\{0,1\right\}$ for all $i$, and $v_j\in \left\{0,1\right\}$ for all $j$.
\end{itemize}
The free energy of the visible layer $\mathbf{v}$ in Eq. \eqref{eq:free_energy_def} can be computed as follows:
\begin{align}
  F(\mathbf{v}) &=-\ln \sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})} \nonumber \\
  &=-\ln \sum_{\mathbf{h}} e^{\mathbf{a}^T \mathbf{v} + \mathbf{b}^T \mathbf{h}
    + \mathbf{h}^T \mathbf{W} \mathbf{v}} \nonumber \\
  &=-\ln \left[e^{\mathbf{a}^T \mathbf{v}} \sum_{\mathbf{h}} e^{\mathbf{b}^T \mathbf{h} 
    + \mathbf{h}^T \mathbf{W} \mathbf{v}}\right] \nonumber \\
  &=-\mathbf{a}^T \mathbf{v} - \ln \sum_{\mathbf{h}} e^{\mathbf{b}^T \mathbf{h} 
    + \mathbf{h}^T \mathbf{W} \mathbf{v}}. 
  \label{eq:free_energy_bernoulli}
\end{align}
Since $h_i$ can take a value $0$ or $1$, 
the sum $\sum_{\mathbf{h}} e^{\mathbf{b}^T \mathbf{h} + \mathbf{h}^T \mathbf{W} \mathbf{v}}$ 
can be computed as follows:
\begin{align}
  \sum_{\mathbf{h}} e^{\mathbf{b}^T \mathbf{h} + \mathbf{h}^T \mathbf{W} \mathbf{v}}
    &=\sum_{\mathbf{h}} e^{\sum_i b_i h_i + \sum_i\sum_j h_i w_{ij} v_j} \nonumber \\
    &=\sum_{\mathbf{h}} e^{\sum_i (b_i + \sum_{j} w_{ij} v_j)h_i} \nonumber \\
    &=\prod_{i} \sum_{\left\{h_i\right\}} e^{(b_i + \mathbf{W}_{i} \mathbf{v})h_i } \nonumber \\
  &= \prod_{i} \left(1 + e^{b_i +  \mathbf{W}_{i} \mathbf{v}}\right),
\end{align}
where $\mathbf{W}_i$ is the $i$-th row of the weight matrix $\mathbf{W}$.
Therefore, the free energy of the visible layer can be expressed as follows:
\begin{align}
  F(\mathbf{v}) &=-\mathbf{a}^T \mathbf{v} - \ln \prod_{i} \left(1 + e^{b_i +  \mathbf{W}_{i} \mathbf{v}}\right) \nonumber \\
  &=-\mathbf{a}^T \mathbf{v} - \sum_i \ln \left(1 + e^{b_i +  \mathbf{W}_{i} \mathbf{v}}\right).
  \label{eq:free_energy_bernoulli_final}
\end{align}

Eqs.\eqref{eq:hidden_cond_prob_rewrite} and \eqref{eq:visible_cond_prob_rewrite}
can be rewritten in the matrix form as follows:
\begin{equation}
  p(\mathbf{h}|\mathbf{v}) = \dfrac{e^{\mathbf{b}^T \mathbf{h} + \mathbf{h}^T \mathbf{W} \mathbf{v}}}
  {\sum_{\mathbf{h}} e^{\mathbf{b}^T \mathbf{h} + \mathbf{h}^T \mathbf{W} \mathbf{v}}}
  \label{eq:hidden_cond_prob_matrix}
\end{equation}
\begin{equation}
  p(\mathbf{v}|\mathbf{h}) = \dfrac{e^{\mathbf{a}^T \mathbf{v} + \mathbf{v}^T \mathbf{W}^T \mathbf{h}}}
  {\sum_{\mathbf{v}} e^{\mathbf{a}^T \mathbf{v} + \mathbf{v}^T \mathbf{W}^T \mathbf{h}}},
  \label{eq:visible_cond_prob_matrix}
\end{equation}
and Eqs.\eqref{eq:hidden_cond_prob_final} and \eqref{eq:visible_cond_prob_final} become:
\begin{equation}
  p(h_j=1|\mathbf{v}) = \dfrac{e^{b_j + \mathbf{W}_j \mathbf{v}}}{1 + e^{b_j + \mathbf{W}_j \mathbf{v}}}
  =\sigma(b_j + \mathbf{W}_j \mathbf{v}),
  \label{eq:hidden_cond_prob_final_matrix}
\end{equation}
\begin{equation}
  p(v_i=1|\mathbf{h}) = \dfrac{e^{a_i + \mathbf{W}^T_i \mathbf{h}}}{1 + e^{a_i + \mathbf{W}^T_i \mathbf{h}}}
  =\sigma(a_i + \mathbf{h}^T\mathbf{W}^T_i ),
  \label{eq:visible_cond_prob_final_matrix}
\end{equation}
where $\sigma(x) = \dfrac{1}{1 + e^{-x}}$ is the sigmoid function.

\subsection{General Log-likelihood and Gradient of the Log-likelihood}
 For a single data point $\mathbf{v}$, the log-likelihood function can be expressed as follows:
\begin{equation}
  \mathcal{L}_{\mathbf{v}} = \ln p(\mathbf{v}) = 
  \ln \left[\dfrac{\sum_{\mathbf{h}} e^{-\epsilon(\mathbf{v}, \mathbf{h})}}{Z}\right]
  = \ln \left[\dfrac{e^{-F(\mathbf{v})}}{Z}\right].
  \label{eq:log_likelihood_free_energy}
\end{equation}

\noindent\textbf{- Parameter Update:} 
\begin{samplebox}[Parameter Update for RBM]
  Let $\theta$ be a parameter of the RBM, which can be a weight $w_{ij}$ or a bias $a_i$ or $b_j$.
  The parameter $\theta$ can be updated using the gradient ascent method as follows:
  \begin{equation}
    \theta \leftarrow \theta + \eta \dfrac{\partial \mathcal{L}}{\partial \theta},
     \label{eq:parameter_update}
  \end{equation}
  where $\eta$ is the learning rate.
\end{samplebox}

\noindent\textbf{-Canculating the gradient of $\mathcal{L}$}\\
\noindent Using Eq.\eqref{eq:log_likelihood_free_energy},
\begin{align}
  \frac{\partial \mathcal{L}}{\partial \theta} 
  &= \dfrac{\partial \mathcal{L}_{\mathbf{v}}}{\partial \theta} \nonumber \\
  &= \dfrac{\partial}{\partial \theta} 
      \ln \left[\dfrac{e^{-F(\mathbf{v})}}{Z}\right] \nonumber \\
  &= \dfrac{\partial}{\partial \theta} \left(-F(\mathbf{v})\right)
      - \dfrac{\partial \ln Z}{\partial \theta} \nonumber \\
  &= -\dfrac{\partial F(\mathbf{v})}{\partial \theta}
      - \dfrac{\partial}{\partial \theta} \ln \left(\sum_{\mathbf{v}} 
      e^{-F(\mathbf{v})}\right) \nonumber\\
  &= -\dfrac{\partial F(\mathbf{v})}{\partial \theta} 
      + \dfrac{\sum_{\mathbf{v}}e^{-F(\mathbf(v))}
      \dfrac{\partial F(\mathbf{v})}{\partial \theta}}
      {\sum_{\mathbf{v}} e^{-F(\mathbf{v})}} \nonumber \\
  &= -\dfrac{\partial}{\partial\theta}F(\mathbf{v})
      +\sum_{\mathbf{v}} \dfrac{e^{-F(\mathbf{v})}}{Z}
      \dfrac{\partial F(\mathbf{v})}{\partial \theta} \nonumber \\
  &= -\dfrac{\partial}{\partial\theta}F(\mathbf{v})
      +\sum_{\mathbf{v}} p(\mathbf{v}) 
      \dfrac{\partial F(\mathbf{v})}{\partial \theta}.
  \label{eq:gradient_theta}
\end{align}
Here, the first term in the last line of Eq.\eqref{eq:gradient_theta} 
can be computed using the training data $\mathbf{v}$,
while the second term in the last line of Eq.\eqref{eq:gradient_theta}
should be computed using the model distribution $p(\mathbf{v})$, i.e.,
it includes summation over all possible $\mathbf{v}$.
Thus, calculating the second term is intractable to compute exactly.
To avoid the difficulty of computing the second term,
we can use the Markov Chain Monte Carlo (MCMC) methods, such as Gibbs sampling,
to approximate the second term.

Here, G.E. Hinton proposed the an efficient CD algorithm, 
which is a practical method for training RBM,
to avoid the difficulty of computing the second term in Eq.\eqref{eq:gradient_theta}.
The CD algorithm approximates the second term in Eq.\eqref{eq:gradient_theta}
using a sample $\mathbf{v}^\prime$ obtained from the Gibbs sampling procedure,
which is much more efficient than the burn-in algorithm.
In the proposed CD algorithm, Hinton showed that even using a single step of Gibbs sampling 
(i.e., $CD_1$) can yield good results in practice,
and using more steps (i.e., $CD_k$ with $k>1$) can further 
improve the performance of the RBM.
In his scheme, the {\it loss} function $L$ is defined as the difference
between the free energy of the training data $\mathbf{v}$
and the free energy of the sample $\mathbf{v}^\prime$ obtained
from the Gibbs sampling procedure, i.e.,
\begin{equation}
L=F(\mathbf{v})-F(\mathbf{v}^\prime).
\label{eq:loss_Hinton}
\end{equation}

\section{Appendix: Implementation of RBM in Python using PyTorch}

#!/usr/bin/env python
# coding: utf-8
"""
RBM 원래 코드 출처: https://github.com/odie2630463/Restricted-Boltzmann-Machines-in-pytorch
공돌이의 수학정리노트 - RBM편

import numpy as np
import torch
import torch.utils.data
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from torch.autograd import Variable
from torchvision import datasets, transforms
from torchvision.utils import make_grid
 
 
#%% 시각화를 위해 필요한 함수
 
# get_ipython().run_line_magic('matplotlib', 'inline') # inline 출력 필요한 경우 주석 처리 제거할 것.
import matplotlib.pyplot as plt
 
def show_adn_save(file_name,img):
    #
    # 이미지 출력해주고 세이브 해주는 역할의 함수
    #
    npimg = np.transpose(img.numpy(),(1,2,0))
    f = "./%s.png" % file_name
    plt.imshow(npimg)
    plt.imsave(f,npimg)
 
 
#%% RBM 모듈 만들기
class RBM(nn.Module):
    def __init__(self,n_vis=784,n_hin=500,k=5):
        super(RBM, self).__init__()
        # 여타 initialization에 비해 이와 같은 방식이 학습이 제일 잘 됨.
        self.W = nn.Parameter(torch.randn(n_hin,n_vis)*1e-2) 
        self.v_bias = nn.Parameter(torch.zeros(n_vis)) # bias는 0으로 초기화.
        self.h_bias = nn.Parameter(torch.zeros(n_hin)) # bias는 0으로 초기화.
        
        self.k = k # Contrastive Divergence의 횟수
    
    def sample_from_p(self,p):
        # Gibbs Sampling 과정
        # 확률 계산 후 uniform distribution으로부터 얻은 값 보다 크면 1, 아니면 -1로 출력하고
        # relu 함수를 이용해 -1로 출력된 값들은 모두 0으로 처리해버림.
        p_ = p - Variable(torch.rand(p.size()))
        p_sign = torch.sign(p_)
        return F.relu(p_sign)
    
    def v_to_h(self,v):
        # 주어진 visible units로부터 hidden unit을 sampling하는 과정
        p_h = torch.sigmoid(F.linear(v,self.W,self.h_bias))
        sample_h = self.sample_from_p(p_h)
        return p_h,sample_h
    
    def h_to_v(self,h):
        # hidden unit의 샘플로부터 visible unit을 다시 복원해보는 과정
        p_v = torch.sigmoid(F.linear(h,self.W.t(),self.v_bias))
        sample_v = self.sample_from_p(p_v)
        return p_v,sample_v
        
    def forward(self,v):
        # RBM에서 forward propagation은 visible -> hidden -> 다시 visible로 
        # hidden unit과 visible unit을 샘플링 하는 과정을 말함.
        pre_h1,h1 = self.v_to_h(v)
        
        h_ = h1
        for _ in range(self.k): # Contrastive Divergence를 k 번 수행함.
            pre_v_,v_ = self.h_to_v(h_)
            pre_h_,h_ = self.v_to_h(v_)
        
        # 아래의 v는 입력으로 들어온 v이고 v_는 sampling으로 얻은 h로부터 다시 획득한 입력 샘플
        return v,v_
    
    def free_energy(self,v):
        # Free Energy 계산
        vbias_term = v.mv(self.v_bias) # mv: matrix - vector product
        wx_b = F.linear(v,self.W,self.h_bias)
        
        temp = torch.log(
            torch.exp(wx_b) + 1
            )
        hidden_term = torch.sum(temp, dim = 1)
        
        return (-hidden_term - vbias_term).mean()
 
 
#%% 데이터 셋 불러오기: MNIST 이용
batch_size = 64
train_loader = torch.utils.data.DataLoader(
    datasets.MNIST('./MNIST_data', train=True, download=True,
                   transform=transforms.Compose([
                       transforms.ToTensor()
                   ])),
    batch_size=batch_size)
 
test_loader = torch.utils.data.DataLoader(
    datasets.MNIST('./MNIST_data', train=False, transform=transforms.Compose([
                       transforms.ToTensor()
                   ])),
    batch_size=batch_size)
 
 
#%% RBM 모델 객체 및 최적화 함수 객체 만들기
 
rbm = RBM(k=1) # CD의 k = 1로 설정. 
# 즉, visible unit을 딱 한번만 sampling 해주고 
# 그걸로 원래의 visible unit의 데이터와 configuration 비교해서 loss로 삼겠다는 의미.
train_op = optim.Adam(rbm.parameters(), 0.005) # 어떤 Optimizer를 넣어줘도 무관함.
 
#%% RBM 학습
 
for epoch in range(10):
    loss_ = []
    for _, (data,target) in enumerate(train_loader): # Stochastic Gradient Descent 수행함.
        data = Variable(data.view(-1,784))
        sample_data = data.bernoulli() # RBM의 입력은 0 또는 1의 값만 가져야 함.
        
        v,v1 = rbm(sample_data)
        loss = rbm.free_energy(v) - rbm.free_energy(v1)
        loss_.append(loss.data.item())
        train_op.zero_grad()
        loss.backward() # RBM의 Backpropagation은 원래의 입력과 sampling 된 입력간의 차이를 줄여주는 과정
        train_op.step()
    
    print(np.mean(loss_))
 
#%% Test Data로부터 원본 데이터와 샘플링된 입력 데이터를 비교해보자.
 
testset = datasets.MNIST('./MNIST_data', train=False, transform=transforms.Compose([
                       transforms.ToTensor()
                   ]))
 
sample_data = testset.data[:32,:].view(-1, 784) # 총 32개 데이터를 받아옴.
sample_data = sample_data.type(torch.FloatTensor)/255.
 
v, v1 = rbm(sample_data)
show_adn_save("real_testdata",make_grid(v.view(32,1,28,28).data))
show_adn_save("generated_testdata",make_grid(v1.view(32,1,28,28).data))
